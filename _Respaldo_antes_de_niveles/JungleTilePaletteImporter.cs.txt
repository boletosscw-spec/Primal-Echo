        private struct SheetPlacement
        {
            public string fileName;
            public int startX;
            public int topY;

            {
                this.fileName = fileName;
                this.startX = startX;
                this.topY = topY;
            }
        }

        static JungleTilePaletteImporter()
        {
            EditorApplication.update += TryAutoImportInEditMode;
            EditorApplication.playModeStateChanged += state =>
            {
                if (state == PlayModeStateChange.EnteredEditMode)
                {
                    EditorApplication.update -= TryAutoImportInEditMode;
                    EditorApplication.update += TryAutoImportInEditMode;
                }
            };
        }

        private static void TryAutoImportInEditMode()
        {
            if (EditorApplication.isPlayingOrWillChangePlaymode || EditorApplication.isCompiling || EditorApplication.isUpdating)
                return;

            EditorApplication.update -= TryAutoImportInEditMode;

            bool alreadyCreated = AssetDatabase.LoadAssetAtPath<TileBase>($"{JungleTilesAssetFolder}/tile_jungle_ground_brown_r0_c0.asset") != null;
            if (!alreadyCreated || EditorPrefs.GetInt(AutoImportKey, 0) < 5)
            {
                EditorPrefs.SetInt(AutoImportKey, 5);
                ImportJungleTilesToPaletteAndBuildKingsPass();
            }
        }

        [MenuItem("Tools/Kael/Construir Mapa Paso del Rey (Hollow Knight) con Jungla")]
        [MenuItem("Primal Echo/Construir Paso del Rey (Hollow Knight) con Tiles de Jungla")]
        public static void ImportJungleTilesToPaletteAndBuildKingsPass()
        {
            if (!AssetDatabase.IsValidFolder("Assets/Tiles"))
            {
                AssetDatabase.CreateFolder("Assets", "Tiles");
            }
            if (!AssetDatabase.IsValidFolder(JungleTilesAssetFolder))
            {
                AssetDatabase.CreateFolder("Assets/Tiles", "Jungle");
            }

            SpriteDataProviderFactories factory = new SpriteDataProviderFactories();
            factory.Init();

            Dictionary<string, Tile> tiles = new Dictionary<string, Tile>();

            // Paso 1: Si aún no están cortados los 17 PNGs en 32x32, cortarlos
            bool needsSlice = AssetDatabase.LoadAssetAtPath<Tile>($"{JungleTilesAssetFolder}/tile_jungle_ground_brown_r1_c1.asset") == null;
            if (needsSlice)
            {
                AssetDatabase.StartAssetEditing();
                try
                {
                    foreach (SheetPlacement sp in SheetPlacements)
                    {
                        string assetPath = $"{JungleSourceFolder}/{sp.fileName}";
                        SliceJungleSheet32x32(assetPath, factory);
                    }
                }
                finally
                {
                    AssetDatabase.StopAssetEditing();
                }
                AssetDatabase.Refresh();

                // Paso 2: Cargar o crear los Tile assets
                AssetDatabase.StartAssetEditing();
                try
                {
                    foreach (SheetPlacement sp in SheetPlacements)
                    {
                        string assetPath = $"{JungleSourceFolder}/{sp.fileName}";
                        string baseName = Path.GetFileNameWithoutExtension(sp.fileName);

                        Object[] allSubAssets = AssetDatabase.LoadAllAssetsAtPath(assetPath);
                        foreach (Object sub in allSubAssets)
                        {
                            if (sub is Sprite sprite && sprite.name.StartsWith(baseName + "_r"))
                            {
                                string tileAssetPath = $"{JungleTilesAssetFolder}/{sprite.name}.asset";
                                Tile tile = AssetDatabase.LoadAssetAtPath<Tile>(tileAssetPath);
                                if (tile == null)
                                {
                                    tile = ScriptableObject.CreateInstance<Tile>();
                                    tile.name = sprite.name;
                                    tile.sprite = sprite;
                                    tile.color = Color.white;
                                    tile.colliderType = Tile.ColliderType.Sprite;
                                    AssetDatabase.CreateAsset(tile, tileAssetPath);
                                }
                                else if (tile.sprite != sprite)
                                {
                                    tile.sprite = sprite;
                                    EditorUtility.SetDirty(tile);
                                }

                                tiles[sprite.name] = tile;
                            }
                        }
                    }
                }
                finally
                {
                    AssetDatabase.StopAssetEditing();
                }

                AssetDatabase.SaveAssets();
                AssetDatabase.Refresh();
            }

            // Asegurar que el diccionario tenga todos los tiles cargados desde Assets/Tiles/Jungle
            string[] tileGuids = AssetDatabase.FindAssets("t:Tile", new[] { JungleTilesAssetFolder });
            foreach (string g in tileGuids)
            {
                string p = AssetDatabase.GUIDToAssetPath(g);
                Tile t = AssetDatabase.LoadAssetAtPath<Tile>(p);
                if (t != null) tiles[t.name] = t;
            }

            // Paso 3: Actualizar las Tile Palettes si hace falta
            if (needsSlice || !File.Exists(JunglePalettePrefabPath))
            {
                PopulatePalettePrefab(MainPalettePrefabPath, tiles, keepExistingTiles: true);
                PopulatePalettePrefab(JunglePalettePrefabPath, tiles, keepExistingTiles: false);
            }

            // Paso 4: Construir el mapa detallado estilo Paso del Rey (Hollow Knight) usando 100% la Tile Palette de Jungla
            BuildKingsPassMap(tiles);

            AssetDatabase.SaveAssets();
            AssetDatabase.Refresh();

            Debug.Log($"<color=#4CAF50><b>[Primal Echo]</b> ¡Mapa 'Paso del Rey' (Hollow Knight) construido al detalle con {tiles.Count} tiles de la Tile Palette de Jungla!</color>");
        }

        private static void SliceJungleSheet32x32(string assetPath, SpriteDataProviderFactories factory)
        {
            if (!File.Exists(assetPath)) return;

            TextureImporter importer = AssetImporter.GetAtPath(assetPath) as TextureImporter;
            if (importer == null) return;

            byte[] bytes = File.ReadAllBytes(assetPath);
            Texture2D rawTex = new Texture2D(2, 2, TextureFormat.RGBA32, false);
            rawTex.LoadImage(bytes);

            int width = rawTex.width;
            int height = rawTex.height;
            int cols = width / CellSizePx;
            int rows = height / CellSizePx;
            Color32[] pixels = rawTex.GetPixels32();
            Object.DestroyImmediate(rawTex);

            importer.textureType = TextureImporterType.Sprite;
            importer.spriteImportMode = SpriteImportMode.Multiple;
            importer.spritePixelsPerUnit = CellSizePx;
            importer.filterMode = FilterMode.Point;
            importer.textureCompression = TextureImporterCompression.Uncompressed;
            importer.alphaIsTransparency = true;
            importer.mipmapEnabled = false;

            string baseName = Path.GetFileNameWithoutExtension(assetPath);
            List<SpriteRect> rects = new List<SpriteRect>();

            for (int r = 0; r < rows; r++)
            {
                int yBottom = height - (r + 1) * CellSizePx;
                for (int c = 0; c < cols; c++)
                {
                    int xLeft = c * CellSizePx;
                    if (!HasVisiblePixels(pixels, width, xLeft, yBottom, CellSizePx, CellSizePx))
                        continue;

                    SpriteRect sr = new SpriteRect
                    {
                        name = $"{baseName}_r{r}_c{c}",
                        rect = new Rect(xLeft, yBottom, CellSizePx, CellSizePx),
                        alignment = SpriteAlignment.Center,
                        pivot = new Vector2(0.5f, 0.5f),
                        spriteID = GUID.Generate()
                    };
                    rects.Add(sr);
                }
            }

            ISpriteEditorDataProvider dataProvider = factory.GetSpriteEditorDataProviderFromObject(importer);
            dataProvider.InitSpriteEditorDataProvider();
            dataProvider.SetSpriteRects(rects.ToArray());
            dataProvider.Apply();

            EditorUtility.SetDirty(importer);
            importer.SaveAndReimport();
        }

        private static bool HasVisiblePixels(Color32[] pixels, int texWidth, int startX, int startY, int w, int h)
        {
            for (int y = startY; y < startY + h; y++)
            {
                int rowOffset = y * texWidth;
                for (int x = startX; x < startX + w; x++)
                {
                    if (pixels[rowOffset + x].a > 5)
                        return true;
                }
            }
            return false;
        }

        private static void PopulatePalettePrefab(string prefabPath, Dictionary<string, Tile> createdTiles, bool keepExistingTiles)
        {
            GameObject paletteRoot = null;
            bool loadedFromExisting = false;

            if (File.Exists(prefabPath))
            {
                paletteRoot = PrefabUtility.LoadPrefabContents(prefabPath);
                loadedFromExisting = true;
            }
            else
            {
                paletteRoot = new GameObject(Path.GetFileNameWithoutExtension(prefabPath), typeof(Grid));
                Grid grid = paletteRoot.GetComponent<Grid>();
                grid.cellSize = new Vector3(1f, 1f, 0f);

                GameObject layerObj = new GameObject("Layer1", typeof(Tilemap), typeof(TilemapRenderer));
                layerObj.transform.SetParent(paletteRoot.transform, false);
            }

            try
            {
                Tilemap paletteMap = paletteRoot.GetComponentInChildren<Tilemap>();
                if (paletteMap != null)
                {
                    if (!keepExistingTiles)
                    {
                        paletteMap.ClearAllTiles();
                    }

                    foreach (SheetPlacement sp in SheetPlacements)
                    {
                        string baseName = Path.GetFileNameWithoutExtension(sp.fileName);
                        int xOffset = keepExistingTiles ? sp.startX : (sp.startX - 21);

                        for (int r = 0; r < 20; r++)
                        {
                            for (int c = 0; c < 20; c++)
                            {
                                string key = $"{baseName}_r{r}_c{c}";
                                if (createdTiles.TryGetValue(key, out Tile tile) && tile != null)
                                {
                                    Vector3Int cellPos = new Vector3Int(xOffset + c, sp.topY - r, 0);
                                    paletteMap.SetTile(cellPos, tile);
                                }
                            }
                        }
                    }

                    paletteMap.CompressBounds();
                }

                GameObject savedPrefab = PrefabUtility.SaveAsPrefabAsset(paletteRoot, prefabPath);
                if (savedPrefab != null && AssetDatabase.LoadAssetAtPath<GridPalette>(prefabPath) == null)
                {
                    GridPalette gridPalette = ScriptableObject.CreateInstance<GridPalette>();
                    gridPalette.name = "Palette Settings";
                    gridPalette.cellSizing = GridPalette.CellSizing.Automatic;
                    AssetDatabase.AddObjectToAsset(gridPalette, savedPrefab);
                    EditorUtility.SetDirty(savedPrefab);
                    AssetDatabase.SaveAssets();
                }
            }
            finally
            {
                if (loadedFromExisting)
                {
                    PrefabUtility.UnloadPrefabContents(paletteRoot);
                }
                else
                {
                    Object.DestroyImmediate(paletteRoot);
                }
            }
        }

        private static void BuildKingsPassMap(Dictionary<string, Tile> tiles)
        {
            Tile T(string name)
            {
                tiles.TryGetValue(name, out Tile res);
                return res;
            }

            // Limpiar props sueltos antiguos para que el 100% del mapa provenga limpiamente de la Tile Palette
            GameObject relics = GameObject.Find("Reliquias_Y_Monumentos") ?? GameObject.Find("Reliquias_Y_Vestigios");
            if (relics != null) Object.DestroyImmediate(relics);
            GameObject props = GameObject.Find("Vegetacion_Y_Props");
            if (props != null) Object.DestroyImmediate(props);
            GameObject oldDeco = GameObject.Find("Agua_Y_Detalles");
            if (oldDeco != null) Object.DestroyImmediate(oldDeco);

            // 1. Configurar Kael en la base de La Caída del Paso del Rey (x = -11, Y = 1.0)
            GameObject player = GameObject.Find("Player");
            if (player != null)
            {
                CapsuleCollider2D pCol = player.GetComponent<CapsuleCollider2D>();
                if (pCol != null)
                {
                    pCol.size = new Vector2(0.45f, 1.35f);
                    pCol.offset = new Vector2(0f, 0.675f);
                    EditorUtility.SetDirty(pCol);
                }
                player.transform.position = new Vector3(-11f, 1.0f, 0f);
                EditorUtility.SetDirty(player.transform);
            }

            // 2. Configurar Cámara (borde inferior exacto en Y = 0.0 para que solo se vea 1 cuadro de piso en la base)
            Camera cam = Camera.main;
            Vector3 camPos = new Vector3(-8f, 6.0f, -10f);
            if (cam != null)
            {
                cam.orthographicSize = 6f;
                cam.backgroundColor = new Color(34f / 255f, 32f / 255f, 52f / 255f, 1f);
                cam.transform.position = camPos;

                CameraFollow2D follow = cam.GetComponent<CameraFollow2D>();
                if (follow != null)
                {
                    follow.SetBounds(new Vector2(-8f, 6.0f), new Vector2(108f, 18f));
                    SerializedObject so = new SerializedObject(follow);
                    SerializedProperty offsetProp = so.FindProperty("offset");
                    if (offsetProp != null)
                    {
                        offsetProp.vector3Value = new Vector3(0f, 5.0f, -10f);
                        so.ApplyModifiedPropertiesWithoutUndo();
                    }
                    EditorUtility.SetDirty(follow);
                }
            }

            // 3. Asegurar que el Fondo Parallax cubra 100% la cámara sin bordes negros
            GameObject bgGroup = GameObject.Find("Fondo_Parallax");
            if (bgGroup != null)
            {
                for (int i = 0; i < bgGroup.transform.childCount; i++)
                {
                    Transform layer = bgGroup.transform.GetChild(i);
                    for (int c = layer.childCount - 1; c >= 0; c--)
                    {
                        Object.DestroyImmediate(layer.GetChild(c).gameObject);
                    }

                    if (layer.name.Contains("01_Panorama") || layer.name.Contains("02_Bosque"))
                    {
                        layer.position = new Vector3(camPos.x, 6.15f, layer.position.z);
                        layer.localScale = new Vector3(1.32f, 1.32f, 1f);
                    }
                    else if (layer.name.Contains("Vegetacion_Cercana"))
                    {
                        layer.position = new Vector3(camPos.x, 0.85f, layer.position.z);
                        layer.localScale = new Vector3(1.22f, 0.95f, 1f);
                    }
                }
            }

            // 4. Obtener o crear los 3 Tilemaps dentro del Grid:
            //    - 'Fondo_Cueva_Y_Arboles' (sortingOrder = -10, sin colisión: árboles grandes, lianas y pilares traseros)
            //    - 'mapa' (sortingOrder = 0, CON colisión: suelo, muros, techos, puentes y ramas caminables)
            //    - 'Decoracion_Sin_Colision' (sortingOrder = 5, sin colisión: puntas de césped, farolas, cristales, tótems, flores, agua y barandillas)
            GameObject gridObj = GameObject.Find("Grid");
            if (gridObj == null) return;

            Tilemap EnsureLayer(string name, int sortingOrder, bool withCollider)
            {
                Transform existing = gridObj.transform.Find(name);
                GameObject go = existing != null ? existing.gameObject : new GameObject(name, typeof(Tilemap), typeof(TilemapRenderer));
                go.transform.SetParent(gridObj.transform, false);
                go.transform.localPosition = Vector3.zero;

                Tilemap tm = go.GetComponent<Tilemap>() ?? go.AddComponent<Tilemap>();
                TilemapRenderer tr = go.GetComponent<TilemapRenderer>() ?? go.AddComponent<TilemapRenderer>();
                tr.sortingOrder = sortingOrder;

                if (withCollider)
                {
                    Rigidbody2D rb = go.GetComponent<Rigidbody2D>() ?? go.AddComponent<Rigidbody2D>();
                    rb.bodyType = RigidbodyType2D.Static;
                    TilemapCollider2D tc = go.GetComponent<TilemapCollider2D>() ?? go.AddComponent<TilemapCollider2D>();
                    tc.compositeOperation = Collider2D.CompositeOperation.Merge;
                    CompositeCollider2D cc = go.GetComponent<CompositeCollider2D>() ?? go.AddComponent<CompositeCollider2D>();
                    cc.geometryType = CompositeCollider2D.GeometryType.Polygons;
                }

                tm.ClearAllTiles();
                return tm;
            }

            Tilemap bgMap = EnsureLayer("Fondo_Cueva_Y_Arboles", -10, false);
            Tilemap solidMap = EnsureLayer("mapa", 0, true);
            Tilemap fgDecoMap = EnsureLayer("Decoracion_Sin_Colision", 5, false);

            // =========================================================================
            // SISTEMA DE AUTO-TILING FIEL AL MOCKUP DE OPP1_JUNGLE_TILES
            // =========================================================================
            const int minX = -20;
            const int maxX = 122;
            const int minY = 0; // ¡NUNCA bajar de y = 0 para que solo se vea 1 cuadro de piso abajo!
            const int maxY = 16;

            int gridW = maxX - minX + 1;
            int gridH = maxY - minY + 1;

            // 0 = vacío, 1 = tierra marrón de jungla, 2 = piedra gris de ruinas
            int[,] cellType = new int[gridW, gridH];
            bool[,] noGrassCap = new bool[gridW, gridH];

            void FillRect(int x1, int x2, int y1, int y2, int type, bool suppressGrass = false)
            {
                x1 = Mathf.Clamp(x1, minX, maxX);
                x2 = Mathf.Clamp(x2, minX, maxX);
                y1 = Mathf.Clamp(y1, minY, maxY);
                y2 = Mathf.Clamp(y2, minY, maxY);
                for (int x = x1; x <= x2; x++)
                {
                    for (int y = y1; y <= y2; y++)
                    {
                        cellType[x - minX, y - minY] = type;
                        if (suppressGrass) noGrassCap[x - minX, y - minY] = true;
                    }
                }
            }

            int GetCell(int x, int y)
            {
                if (x < minX || x > maxX) return 2;
                if (y < minY) return 1; // Por debajo de y=0 se considera macizo para que y=0 no pinte borde de techo sino piso
                if (y > maxY) return 0;
                return cellType[x - minX, y - minY];
            }

            // =========================================================================
            // DISEÑO DE NIVEL: PASO DEL REY (KING'S PASS - HOLLOW KNIGHT)
            // =========================================================================

            // --- ZONA 1: La Caída del Abismo (x = -20 a 5) ---
            FillRect(-20, -16, 0, 13, 2, suppressGrass: true);
            FillRect(-15, 5, 0, 0, 1);

            // --- ZONA 2: Las Primeras Puertas y Escalones de Piedra (x = 6 a 26) ---
            FillRect(6, 26, 0, 0, 1);
            FillRect(7, 16, 1, 1, 1);
            FillRect(11, 15, 2, 2, 1);

            // --- ZONA 3: El Puente Colgante y la Laguna Subterránea (x = 27 a 48) ---
            FillRect(27, 30, 0, 2, 1);
            FillRect(37, 43, 0, 2, 1);
            FillRect(44, 46, 0, 1, 1);
            FillRect(47, 48, 0, 0, 1);

            // --- ZONA 4: La Caverna Vertical del Árbol y el Santuario Secreto (x = 49 a 75) ---
            FillRect(49, 75, 0, 0, 1);
            FillRect(66, 75, 1, 1, 1);
            FillRect(69, 75, 2, 2, 2);
            FillRect(71, 75, 3, 3, 2);

            // --- ZONA 5: El Abismo de los Pilares sobre la Laguna (x = 76 a 95) ---
            FillRect(79, 81, 0, 2, 2);
            FillRect(85, 87, 0, 3, 2);
            FillRect(91, 93, 0, 2, 2);

            // --- ZONA 6: Ascenso al Balcón de Bocasucia / Mirador Final (x = 96 a 122) ---
            FillRect(96, 122, 0, 1, 1);
            FillRect(98, 122, 2, 2, 1);
            FillRect(101, 122, 3, 3, 2);
            FillRect(104, 122, 4, 4, 2);
            FillRect(118, 122, 5, 13, 2, suppressGrass: true);

            // =========================================================================
            // PINTADO AUTOMÁTICO DEL TERRENO SÓLIDO Y PUNTAS DE CÉSPED
            // =========================================================================
            Tile[] brownLeftWall = { T("tile_jungle_wall_brown_r0_c0"), T("tile_jungle_wall_brown_r1_c0"), T("tile_jungle_wall_brown_r2_c0"), T("tile_jungle_wall_brown_r3_c0") };
            Tile[] brownRightWall = { T("tile_jungle_wall_brown_r0_c1"), T("tile_jungle_wall_brown_r1_c1"), T("tile_jungle_wall_brown_r2_c1"), T("tile_jungle_wall_brown_r3_c1") };
            Tile[] greyLeftWall = { T("tile_jungle_wall_grey_r0_c0"), T("tile_jungle_wall_grey_r1_c0"), T("tile_jungle_wall_grey_r2_c0"), T("tile_jungle_wall_grey_r3_c0") };
            Tile[] greyRightWall = { T("tile_jungle_wall_grey_r0_c1"), T("tile_jungle_wall_grey_r1_c1"), T("tile_jungle_wall_grey_r2_c1"), T("tile_jungle_wall_grey_r3_c1") };

            Tile solidDarkFill = T("tile_jungle_wall_brown_r0_c2");
            Tile[] subtleFillAccents = {
                T("tile_jungle_wall_brown_r1_c2"),
                T("tile_jungle_wall_brown_r2_c3"),
                T("tile_jungle_wall_grey_r1_c2")
            };

            for (int x = minX; x <= maxX; x++)
            {
                for (int y = minY; y <= maxY; y++)
                {
                    int type = GetCell(x, y);
                    if (type == 0) continue;

                    bool isGrey = (type == 2);
                    string gPrefix = isGrey ? "tile_jungle_ground_grey" : "tile_jungle_ground_brown";
                    string bPrefix = isGrey ? "tile_jungle_bottom_grey" : "tile_jungle_bottom_brown";

                    bool up = GetCell(x, y + 1) != 0;
                    bool down = GetCell(x, y - 1) != 0;
                    bool left = GetCell(x - 1, y) != 0;
                    bool right = GetCell(x + 1, y) != 0;

                    Tile chosen = solidDarkFill;

                    if (!up)
                    {
                        // Superficie superior con césped
                        int variantRow = (Mathf.Abs(x) % 3 == 0) ? 3 : 1;
                        if (!left) chosen = T($"{gPrefix}_r{variantRow}_c0") ?? T($"{gPrefix}_r1_c0");
                        else if (!right) chosen = T($"{gPrefix}_r{variantRow}_c2") ?? T($"{gPrefix}_r1_c2");
                        else chosen = T($"{gPrefix}_r{variantRow}_c1") ?? T($"{gPrefix}_r1_c1");

                        // Pintar las puntas de césped (r0 o r2) en la capa decorativa (y + 1) si el bloque superior está libre
                        if (!noGrassCap[x - minX, y - minY] && y + 1 <= maxY && GetCell(x, y + 1) == 0)
                        {
                            int capRow = variantRow - 1;
                            Tile capTile;
                            if (!left) capTile = T($"{gPrefix}_r{capRow}_c0") ?? T($"{gPrefix}_r0_c0");
                            else if (!right) capTile = T($"{gPrefix}_r{capRow}_c2") ?? T($"{gPrefix}_r0_c2");
                            else capTile = T($"{gPrefix}_r{capRow}_c1") ?? T($"{gPrefix}_r0_c1");

                            if (capTile != null)
                            {
                                fgDecoMap.SetTile(new Vector3Int(x, y + 1, 0), capTile);
                            }
                        }
                    }
                    else if (!down && y > minY)
                    {
                        // Borde inferior de techos y plataformas suspendidas
                        if (!left) chosen = T($"{bPrefix}_r0_c0");
                        else if (!right) chosen = T($"{bPrefix}_r0_c3");
                        else chosen = T($"{bPrefix}_r0_c{(Mathf.Abs(x) % 2) + 1}");
                    }
                    else if (!left)
                    {
                        Tile[] wallArr = isGrey ? greyLeftWall : brownLeftWall;
                        chosen = wallArr[Mathf.Abs(y) % wallArr.Length] ?? solidDarkFill;
                    }
                    else if (!right)
                    {
                        Tile[] wallArr = isGrey ? greyRightWall : brownRightWall;
                        chosen = wallArr[Mathf.Abs(y) % wallArr.Length] ?? solidDarkFill;
                    }
                    else
                    {
                        int h = Mathf.Abs(x * 31 + y * 17);
                        if (h % 11 == 0 && subtleFillAccents.Length > 0)
                            chosen = subtleFillAccents[h % subtleFillAccents.Length] ?? solidDarkFill;
                        else
                            chosen = solidDarkFill;
                    }

                    if (chosen != null)
                    {
                        solidMap.SetTile(new Vector3Int(x, y, 0), chosen);
                    }
                }
            }

            // =========================================================================
            // DETALLES ICÓNICOS DE KING'S PASS CON LA TILE PALETTE DE JUNGLA
            // =========================================================================

            // Helper para estampar un bloque rectangular de tiles de una hoja en un Tilemap
            void StampBlock(Tilemap targetMap, string sheetPrefix, int startX, int bottomY, int rStart, int rEnd, int cStart, int cEnd)
            {
                int heightRows = rEnd - rStart + 1;
                for (int r = rStart; r <= rEnd; r++)
                {
                    int ty = bottomY + (heightRows - 1 - (r - rStart));
                    for (int c = cStart; c <= cEnd; c++)
                    {
                        int tx = startX + (c - cStart);
                        Tile t = T($"{sheetPrefix}_r{r}_c{c}");
                        if (t != null)
                        {
                            targetMap.SetTile(new Vector3Int(tx, ty, 0), t);
                        }
                    }
                }
            }

            // Plataformas flotantes de piedra con base rocosa limpia (1 tile de grosor + puntas de césped encima)
            void PlaceFloatingStonePlatform(int startX, int length, int y, bool isGrey = true)
            {
                string gPrefix = isGrey ? "tile_jungle_ground_grey" : "tile_jungle_ground_brown";
                for (int i = 0; i < length; i++)
                {
                    int x = startX + i;
                    int col = (i == 0) ? 0 : (i == length - 1 ? 2 : 1);
                    Tile body = T($"{gPrefix}_r3_c{col}") ?? T($"{gPrefix}_r1_c{col}");
                    Tile cap = T($"{gPrefix}_r2_c{col}") ?? T($"{gPrefix}_r0_c{col}");
                    if (body != null) solidMap.SetTile(new Vector3Int(x, y, 0), body);
                    if (cap != null) fgDecoMap.SetTile(new Vector3Int(x, y + 1, 0), cap);
                }
            }

            // Plataformas escalonadas de ascenso al santuario secreto en la Zona 4
            PlaceFloatingStonePlatform(50, 5, 3, true);
            PlaceFloatingStonePlatform(45, 6, 6, true);

            // --- PUENTE COLGANTE DE CUERDA (Zona 3: x = 31 a 36, piso caminable en y = 2) ---
            for (int i = 0; i < 6; i++)
            {
                int bx = 31 + i;
                Tile rail = T($"tile_jungle_bridge_r0_c{i}");
                Tile plank = T($"tile_jungle_bridge_r1_c{i}");
                Tile ropeUnder = T($"tile_jungle_bridge_r2_c{i}");

                if (rail != null) fgDecoMap.SetTile(new Vector3Int(bx, 3, 0), rail);
                if (plank != null) solidMap.SetTile(new Vector3Int(bx, 2, 0), plank);
                if (ropeUnder != null) fgDecoMap.SetTile(new Vector3Int(bx, 1, 0), ropeUnder);
            }

            // --- AGUA SUBTERRÁNEA PLANA Y CONTINUA (c2 y c4 en r1 son superficie; r2..r3 son agua profunda) ---
            void FillWaterBasin(int xStart, int xEnd, int surfaceY)
            {
                for (int x = xStart; x <= xEnd; x++)
                {
                    int surfCol = (Mathf.Abs(x) % 2 == 0) ? 2 : 4;
                    int deepRow = 2 + (Mathf.Abs(x) % 2);
                    int deepCol = 2 + ((Mathf.Abs(x) / 2) % 2);

                    if (GetCell(x, surfaceY) == 0)
                    {
                        Tile surf = T($"tile_jungle_water_r1_c{surfCol}");
                        if (surf != null) fgDecoMap.SetTile(new Vector3Int(x, surfaceY, 0), surf);
                    }
                    if (surfaceY - 1 >= 0 && GetCell(x, surfaceY - 1) == 0)
                    {
                        Tile deep = T($"tile_jungle_water_r{deepRow}_c{deepCol}");
                        if (deep != null) fgDecoMap.SetTile(new Vector3Int(x, surfaceY - 1, 0), deep);
                    }
                }
            }

            FillWaterBasin(31, 36, 1);
            FillWaterBasin(76, 95, 1);

            // --- RAMAS CAMINABLES MUSGOSAS (Zona 3 sobre el puente y Zona 4 junto al gran árbol) ---
            void PlaceWalkableBranch(int startX, int walkY)
            {
                for (int i = 0; i < 6; i++)
                {
                    int c = 4 + i;
                    int x = startX + i;
                    Tile topCap = T($"tile_jungle_treelimb_r0_c{c}");
                    Tile walkBody = T($"tile_jungle_treelimb_r1_c{c}");
                    Tile hangMoss = T($"tile_jungle_treelimb_r2_c{c}");

                    if (topCap != null) fgDecoMap.SetTile(new Vector3Int(x, walkY + 1, 0), topCap);
                    if (walkBody != null) solidMap.SetTile(new Vector3Int(x, walkY, 0), walkBody);
                    if (hangMoss != null) fgDecoMap.SetTile(new Vector3Int(x, walkY - 1, 0), hangMoss);
                }
            }

            PlaceWalkableBranch(31, 5);
            PlaceWalkableBranch(60, 5);

            // --- GRANDES ÁRBOLES ANCESTRALES (solo tronco limpio cols 0..5 + follaje en las ramas cols 6..7) ---
            void PlaceCleanTree(string prefix, int baseX, int baseY)
            {
                StampBlock(bgMap, prefix, baseX, baseY, 3, 10, 0, 5);
                StampBlock(bgMap, prefix, baseX + 1, baseY + 4, 6, 7, 6, 7);
                StampBlock(bgMap, prefix, baseX + 3, baseY + 3, 6, 7, 6, 7);
            }

            PlaceCleanTree("tile_jungle_tree_light", 55, 1);
            PlaceCleanTree("tile_jungle_tree_dark", 109, 5);

            // --- OBJETOS DETALLADOS DE LA HOJA 'tile_jungle_plants_objects' ---
            // 1. Farolas de Luciérnagas (1x2: r10_c5, r11_c5)
            void PlaceLantern(int x, int groundSurfaceY)
            {
                StampBlock(fgDecoMap, "tile_jungle_plants_objects", x, groundSurfaceY, 10, 11, 5, 5);
            }

            PlaceLantern(-7, 1);   // Farola de bienvenida tras caer a King's Pass
            PlaceLantern(25, 1);   // Farola antes del puente colgante
            PlaceLantern(50, 1);   // Farola al pie de la caverna vertical
            PlaceLantern(74, 4);   // Farola antes del abismo de los pilares
            PlaceLantern(105, 5);  // Farola de entrada al balcón final
            PlaceLantern(114, 5);  // Farola junto a la meta

            // 2. Tótems / Tablillas de Lore Ancestral (2x2: r11..12, c12..13 y c14..15)
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", -4, 1, 11, 12, 12, 13);  // Primera Tablilla de Lore (Inicio)
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 46, 7, 11, 12, 14, 15);  // Tablilla secreta en el balcón superior
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 107, 5, 11, 12, 14, 15); // Tablilla final del Mirador

            // 3. Montículos de Cristales Brillantes (2x2: Azul c6..7, Verde c8..9, Rosa c10..11)
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 12, 3, 11, 12, 6, 7);    // Cristal Azul sobre el escalón
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 39, 3, 11, 12, 8, 9);    // Cristal Verde tras cruzar el puente
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 49, 7, 11, 12, 10, 11);  // Cristal Rosa en el santuario secreto alto
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 52, 4, 11, 12, 6, 7);    // Cristal Azul en la plataforma media
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 116, 5, 11, 12, 8, 9);   // Cristal Verde en la cumbre

            // 4. Tienda de Campaña del Explorador (3x2: r11..12, c1..3) y Troncos Huecos (2x1: r9, c8..9)
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 20, 1, 11, 12, 1, 3);
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 2, 1, 9, 9, 8, 9);
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 67, 2, 10, 10, 8, 9);

            // 5. Estandartes de Calavera (1x2: r10..11, c4) advirtiendo de los abismos
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 29, 3, 10, 11, 4, 4);
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 75, 4, 10, 11, 4, 4);

            // 6. Plantas Exóticas, Helechos, Hongos y Flores de Cueva
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 8, 2, 7, 7, 8, 9);       // Helecho bajo
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 17, 1, 2, 3, 10, 11);    // Planta roja alta
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 41, 3, 2, 3, 8, 9);      // Flor roja de jungla
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 63, 1, 0, 1, 8, 9);      // Bulbo violáceo de cueva
            StampBlock(fgDecoMap, "tile_jungle_plants_objects", 99, 3, 7, 7, 8, 9);      // Helecho en el ascenso final

            // Hongos (r9_c10, r9_c11) y Flores luminosas (r10_c6, r10_c7)
            int[] flowerCols = { -13, -9, -1, 5, 15, 24, 38, 45, 53, 68, 80, 86, 92, 102, 111 };
            for (int i = 0; i < flowerCols.Length; i++)
            {
                int fx = flowerCols[i];
                // Buscar la altura de la superficie en fx
                int topY = -1;
                for (int y = maxY; y >= minY; y--)
                {
                    if (GetCell(fx, y) != 0) { topY = y; break; }
                }
                if (topY >= 0)
                {
                    string itemName = (i % 4 == 0) ? "tile_jungle_plants_objects_r10_c7"
                                    : (i % 4 == 1) ? "tile_jungle_plants_objects_r10_c6"
                                    : (i % 4 == 2) ? "tile_jungle_plants_objects_r9_c10"
                                    : "tile_jungle_plants_objects_r9_c11";
                    Tile itemTile = T(itemName);
                    if (itemTile != null)
                    {
                        fgDecoMap.SetTile(new Vector3Int(fx, topY + 1, 0), itemTile);
                    }
                }
            }

            // 7. Lianas Colgantes desde las Bóvedas (usando 'tile_jungle_bg_vines_r6_c6' vertical)
            Tile vineSegment = T("tile_jungle_bg_vines_r6_c6") ?? T("tile_jungle_bg_vines_r4_c0");
            if (vineSegment != null)
            {
                int[,] vines = {
                    { -12, 10, 4 }, { -9, 10, 3 },
                    { 10, 7, 3 }, { 18, 7, 4 },
                    { 77, 9, 4 }, { 83, 9, 5 }, { 88, 8, 3 }, { 93, 9, 4 }
                };
                for (int i = 0; i < vines.GetLength(0); i++)
                {
                    int vx = vines[i, 0];
                    int vyTop = vines[i, 1];
                    int len = vines[i, 2];
                    for (int k = 0; k < len; k++)
                    {
                        bgMap.SetTile(new Vector3Int(vx, vyTop - k, 0), vineSegment);
                    }
                }
            }

            // 8. Configurar Iluminación 2D Atmosférica (en cada farola y cristal)
            GameObject lightRoot = GameObject.Find("=== ILUMINACION ===");
            if (lightRoot != null)
            {
                for (int i = lightRoot.transform.childCount - 1; i >= 0; i--)
                {
                    Transform child = lightRoot.transform.GetChild(i);
                    if (child.name != "Luz_Global_2D")
                    {
                        Object.DestroyImmediate(child.gameObject);
                    }
                }

                void AddGlowLight(string name, Vector3 pos, Color color, float intensity, float radius)
                {
                    GameObject lObj = new GameObject(name, typeof(Light2D), typeof(AtmosphereGlow));
                    lObj.transform.SetParent(lightRoot.transform, false);
                    lObj.transform.position = pos;
                    Light2D l = lObj.GetComponent<Light2D>();
                    l.lightType = Light2D.LightType.Point;
                    l.color = color;
                    l.intensity = intensity;
                    l.pointLightOuterRadius = radius;
                    l.pointLightInnerRadius = radius * 0.25f;
                    lObj.GetComponent<AtmosphereGlow>().SetGlow(intensity * 0.8f, intensity * 1.2f, 1.8f);
                }

                Color warmLantern = new Color(1f, 0.82f, 0.45f, 1f);
                Color cyanCrystal = new Color(0.3f, 0.92f, 1f, 1f);
                Color pinkCrystal = new Color(1f, 0.45f, 0.85f, 1f);

                AddGlowLight("Luz_Farola_Inicio", new Vector3(-6.5f, 2.5f, 0f), warmLantern, 1.4f, 5.5f);
                AddGlowLight("Luz_Cristal_Escalon", new Vector3(14f, 4.0f, 0f), cyanCrystal, 1.3f, 4.8f);
                AddGlowLight("Luz_Farola_Puente", new Vector3(25.5f, 2.5f, 0f), warmLantern, 1.3f, 5.0f);
                AddGlowLight("Luz_Cristal_Secreto", new Vector3(53f, 10.0f, 0f), pinkCrystal, 1.5f, 5.5f);
                AddGlowLight("Luz_Farola_Abismo", new Vector3(74.5f, 5.5f, 0f), warmLantern, 1.4f, 5.2f);
                AddGlowLight("Luz_Cumbre_Bocasucia", new Vector3(108f, 6.5f, 0f), cyanCrystal, 1.8f, 7.5f);
            }

            // 9. Colocar la Meta del Nivel en el Balcón Final (x = 108, Y = 6.5)
            GameObject goalObj = GameObject.Find("Meta_Mirador_Nivel1");
            if (goalObj != null)
            {
                goalObj.transform.position = new Vector3(108f, 6.5f, 0f);
                EditorUtility.SetDirty(goalObj.transform);
            }

            var scene = EditorSceneManager.GetActiveScene();
            if (scene.IsValid())
            {
                EditorSceneManager.MarkSceneDirty(scene);
                EditorSceneManager.SaveScene(scene);
            }
        }
    }
